# METHOD internal/x is /v1/internal/x; a bare name (channel.list) is POST /v1/<name>. Errors: {code, message}.
# {id}, channel_id: an id's unique start, or self (your session); guild_id=self: your group. person only: not an agent.
# message.create: _data.text (plain words, escaped by the server) for content; _data.paths: local files (socket only).
# A path or folder: ~/x, app:path at an app's computer, host:path; a relative one goes from your session's folder.

# common
channel.create {_data.agent, _data.cwd, _data.text?, _data.model?, _data.mode?, _data.effort?, _data.preset?, data.name?, data.parent_id?, guild_id?} -> Channel  start a session, a child of yours; _data.cwd=. your folder; --wait for its reply
  data.parent_id: the session it hangs under on the page (its launcher). Not given: the session your ir runs in, so one you start is your child: listed under you, archived with you. To start one of its own: channel.update data.parent_id:=null after.
  _data.cwd: . is your own folder and a relative path goes from it; ~ and ~/x the home; host:path here; app:path at that app's computer (GET internal/apps), where it then runs; an absolute path as it is.
  guild_id: a folder group's id (guild.list), self yours; not given: its folder's group, yours when yours holds its folder.
  _data.agent: claude, codex, pi or dsh (GET internal/agents: those that can start here, their modes); _data.mode: one of them; model, effort, preset: the agent's own (SessionState.models, efforts; GET internal/sessions/{id}/presets).
  _data.text: its first message, from you: it reads [ir:<your session>] before it. ir prints the id, with text once the agent took it; --wait: once that turn ended, its reply too; later: GET internal/sessions/{id} --wait.
channel.list {guild_id?, next?, _data.all?} -> List<Channel>  the folders (type 2), then the sessions; _data.all adds archived  view=id,name,_data.status,_data.agent,_data.cwd,_data.title
  parent_id: what it is under: the session it hangs under (its launcher, a subagent's session), else its folder (folder:<path>); --pick id,name,parent_id shows who is under whom.
  _data: a session's Session (_data.own: a person opened it or took it out); a folder's path, color, pinned, order. guild_id: a folder group's (guild.list); self: yours.
channel.get {channel_id} -> Channel  one session, or a folder (folder:<path>); _data is its Session  view=id,name,_data.status,_data.cwd,_data.title,_data.last.text
channel.update {channel_id, data.name?, data.parent_id?, _data.archived?, _data.pinned?, _data.color?, _data.order?, _data.workspace?} -> {}  rename (not over a person's name), archive, restore, pin, tag, move
  data.name: a short name; an agent's never covers one a person gave. _data.archived: true archives it and the sessions hung under it (their agents stop), false brings them back.
  data.parent_id:=null (or its folder): out from under its launcher, into its own folder's group, for good. _data.workspace: into that folder group (guild.list) by hand, 0 none.
  _data.color: red, orange, yellow, green, blue, purple, "" none; pinned; order: the key it was dragged to among those beside it. A folder (folder:<path>) takes color (one here), pinned, order.
channel.delete {channel_id} -> {}  archive: its agent stops, the transcript stays
  The sessions hung under it are archived with it; channel.update _data.archived:=false brings them back.
guild.list {next?} -> List<Guild>  folder groups, then loose  view=id,name,_data.path
message.list {channel_id, next?, direction?, limit?, order?, _data.blocks?} -> BidiList<Message>  transcript; direction before|after|around  view=id,content
  _data.blocks:=true: the page's blocks as Iridium keeps them too, in _data.blocks (kind, text, from, tool calls).
message.get {channel_id, message_id} -> Message  one block, untruncated
message.update {channel_id, message_id, content?, _data.text?} -> {}  change a message you sent while it waits in the queue (message_id: message.create's id)
message.delete {channel_id, message_id} -> {}  take back a message waiting in the queue
message.create {channel_id, content?, _data.text?, _data.paths?, _data.attachments?, _data.steer?} -> Message[]  send; joins a running turn (a person may queue: _data.steer=false)  view=id,_data.queued
  _data.text: plain words (the server escapes them); content: Satori elements as they are. The session reads [ir:<your session>] before it.
  _data.paths: files here to attach (relative: from your folder; socket only); _data.attachments: refs upload.create gave.
  Returns once the agent took it; --wait: once that turn ended, with its reply (exit 2: it waits for a person).
POST internal/links {path} -> FileLink  a file's ir: link and the Markdown to put it in your reply; socket only  view=markdown
PUT internal/secrets/{name} {note?, value?, scope?, workspaces?, always?} -> SecretInfo  set; an agent gets 202 {need}: --wait for the user to type it in
GET internal/sessions/{id} - -> SessionState  live state; --wait: wait for the turn to end, print its reply  view=id,status,model,mode,error,asks,queue,end
  --wait: until its turn ends, then its last reply; exit 2 when it waits for a person (asks: its questions; POST internal/button answers one).
POST internal/sessions/{id}/interrupt - -> {ok}  end the current turn (Esc); the queue pauses
POST internal/sessions/{id}/stop - -> {ok}  stop the agent process; the next message restarts it
POST internal/button {id, option?, choices?, words?, message?} -> {}  answer an ask: id its card (asks[].card.id); or press a button (<card>#<option>; one attached: att:<session>:<from>:<id>, SessionState.attached)
POST internal/files/copy {from, to} -> ndjson  copy a file: from, to are paths here or app:path at an app's computer; prints path, size, sha256; run again to resume; socket only
POST internal/machines/{id}/shell {line, cwd?} -> output  a command line at an app's computer (id or name); --wait for its output and exit code; host or agents
PATCH internal/guilds/{id} {name?, position?, split?} -> {ok}  rename a folder group, move it (0 first), split it by subfolder; host or web
GET internal/apps - -> Apps  apps, computers, logins  view=apps.id,apps.name,apps.online,apps.computer.home,logins.id,logins.name
GET internal/files?path&machine - -> DirListing  list a directory; socket only  view=entries.name,entries.size,entries.type
GET internal/usage?days&app&by - -> Usage  tokens, cost; app=id|name|me, by=hour|weekhour  view=rows.day,rows.agent,rows.model,rows.output,rows.costUsd
